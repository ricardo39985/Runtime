cmake_minimum_required(VERSION 3.25)
project(runtime VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RUNTIME_BUILD_SERVER "Build Drogon/PostgreSQL server" ON)
option(RUNTIME_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(runtime_domain INTERFACE)
target_include_directories(runtime_domain INTERFACE services/core/include)
target_compile_options(runtime_domain INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(RUNTIME_SANITIZERS)
 target_compile_options(runtime_domain INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(runtime_domain INTERFACE -fsanitize=address,undefined)
endif()
find_package(OpenSSL REQUIRED)
find_path(BOOST_JSON_INCLUDE boost/json.hpp REQUIRED)
find_path(ZSTD_INCLUDE zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd zstd_static libzstd_static REQUIRED)
add_library(runtime_applications services/core/src/application.cpp services/core/src/storage.cpp services/core/src/json.cpp)
target_include_directories(runtime_applications SYSTEM PUBLIC ${BOOST_JSON_INCLUDE} ${ZSTD_INCLUDE})
target_link_libraries(runtime_applications PUBLIC runtime_domain OpenSSL::Crypto ${ZSTD_LIBRARY})
include(CTest)
if(BUILD_TESTING)
 add_executable(domain_tests tests/unit/domain_test.cpp)
 target_link_libraries(domain_tests PRIVATE runtime_domain)
 add_test(NAME domain COMMAND domain_tests)
 add_executable(application_tests tests/unit/application_test.cpp)
 target_link_libraries(application_tests PRIVATE runtime_applications)
 add_test(NAME applications COMMAND application_tests)
endif()
if(RUNTIME_BUILD_SERVER)
 find_package(Drogon CONFIG REQUIRED)
 find_package(PostgreSQL REQUIRED)
 add_executable(runtime-core services/core/src/apps_main.cpp)
 target_link_libraries(runtime-core PRIVATE runtime_applications Drogon::Drogon PostgreSQL::PostgreSQL)
 install(TARGETS runtime-core RUNTIME DESTINATION bin)
endif()
