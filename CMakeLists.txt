cmake_minimum_required(VERSION 3.25)
project(runtime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RUNTIME_BUILD_SERVER "Build Drogon/PostgreSQL server" ON)
option(RUNTIME_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(runtime_domain INTERFACE)
target_include_directories(runtime_domain INTERFACE services/core/include)
target_compile_options(runtime_domain INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(RUNTIME_SANITIZERS)
  target_compile_options(runtime_domain INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(runtime_domain INTERFACE -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(domain_tests tests/unit/domain_test.cpp)
  target_link_libraries(domain_tests PRIVATE runtime_domain)
  add_test(NAME domain COMMAND domain_tests)
endif()
if(RUNTIME_BUILD_SERVER)
  find_package(Drogon CONFIG REQUIRED)
  find_package(nlohmann_json CONFIG REQUIRED)
  find_package(nlohmann_json_schema_validator CONFIG REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(PostgreSQL REQUIRED)
  add_executable(runtime-core services/core/src/main.cpp)
  target_link_libraries(runtime-core PRIVATE runtime_domain Drogon::Drogon nlohmann_json::nlohmann_json nlohmann_json_schema_validator::validator OpenSSL::Crypto PostgreSQL::PostgreSQL)
  install(TARGETS runtime-core RUNTIME DESTINATION bin)
endif()
