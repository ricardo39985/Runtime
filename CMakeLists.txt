cmake_minimum_required(VERSION 3.25)
project(runtime VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RUNTIME_BUILD_SERVER "Build the general application server" ON)
option(RUNTIME_STORAGE_TESTS "Build lossless storage tests" ON)
option(RUNTIME_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(runtime_domain INTERFACE)
target_include_directories(runtime_domain INTERFACE services/core/include)
target_compile_options(runtime_domain INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(RUNTIME_SANITIZERS)
 target_compile_options(runtime_domain INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(runtime_domain INTERFACE -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
 foreach(name IN ITEMS domain application generation_budget)
  add_executable(${name}_tests tests/unit/${name}_test.cpp)
  target_link_libraries(${name}_tests PRIVATE runtime_domain)
  add_test(NAME ${name} COMMAND ${name}_tests)
 endforeach()
endif()
if(RUNTIME_STORAGE_TESTS OR RUNTIME_BUILD_SERVER)
 find_package(OpenSSL REQUIRED)
 find_package(zstd CONFIG QUIET)
 add_library(runtime_storage INTERFACE)
 target_link_libraries(runtime_storage INTERFACE runtime_domain OpenSSL::Crypto)
 if(TARGET zstd::libzstd_static)
  target_link_libraries(runtime_storage INTERFACE zstd::libzstd_static)
 elseif(TARGET zstd::libzstd_shared)
  target_link_libraries(runtime_storage INTERFACE zstd::libzstd_shared)
 else()
  find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
  find_library(ZSTD_LIBRARY zstd REQUIRED)
  target_include_directories(runtime_storage INTERFACE ${ZSTD_INCLUDE_DIR})
  target_link_libraries(runtime_storage INTERFACE ${ZSTD_LIBRARY})
 endif()
endif()
if(BUILD_TESTING AND RUNTIME_STORAGE_TESTS)
 add_executable(storage_tests tests/unit/storage_test.cpp)
 target_link_libraries(storage_tests PRIVATE runtime_storage)
 add_test(NAME storage COMMAND storage_tests)
endif()
if(RUNTIME_BUILD_SERVER)
 find_package(Drogon CONFIG REQUIRED)
 find_package(nlohmann_json CONFIG REQUIRED)
 find_package(PostgreSQL REQUIRED)
 add_executable(runtime-core services/core/src/studio_main.cpp)
 target_link_libraries(runtime-core PRIVATE runtime_storage Drogon::Drogon nlohmann_json::nlohmann_json PostgreSQL::PostgreSQL)
 if(BUILD_TESTING)
  add_executable(application_json_tests tests/unit/application_json_test.cpp)
  target_link_libraries(application_json_tests PRIVATE runtime_domain nlohmann_json::nlohmann_json)
  add_test(NAME application_json COMMAND application_json_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples/creative-studio.json ${CMAKE_CURRENT_SOURCE_DIR}/examples/editorial-desk.json)
 endif()
 install(TARGETS runtime-core RUNTIME DESTINATION bin)
endif()
